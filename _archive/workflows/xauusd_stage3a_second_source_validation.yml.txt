name: XAUUSD Stage 3A Second-Source Validation

on:
  workflow_dispatch:
    inputs:
      include_run_link:
        description: "Include GitHub run link in Telegram success messages"
        required: true
        default: "false"

permissions:
  contents: read

concurrency:
  group: xauusd-stage3a-second-source-validation
  cancel-in-progress: true

jobs:
  stage3a-second-source-validation:
    runs-on: ubuntu-latest

    env:
      TELEGRAM_BOT_TOKEN: ${{ secrets.TELEGRAM_BOT_TOKEN }}
      TELEGRAM_CHAT_ID: ${{ secrets.TELEGRAM_CHAT_ID }}
      TELEGRAM_INCLUDE_RUN_LINK: ${{ github.event.inputs.include_run_link }}

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Verify primary SQLite store exists
        run: |
          test -f data/store/xauusd.sqlite

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: "pip"
          cache-dependency-path: requirements.txt

      - name: Install dependencies
        run: |
          python -m pip install -r requirements.txt

      - name: Run Stage 3A second-source validation
        run: |
          python -m app.xauusd_stage3a_second_source_validate

      - name: Send Telegram notification
        if: always()
        env:
          WORKFLOW_STATUS: ${{ job.status }}
        run: |
          python -m app.telegram_notify --status "${WORKFLOW_STATUS}"

      - name: Upload Stage 3A artifacts
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: xauusd-stage3a-second-source-validation-${{ github.run_number }}
          path: |
            data/reports/stage3a_second_source_summary_*.json
            data/reports/stage3a_second_source_summary_*.md
            data/second_source/manifest.json
          if-no-files-found: warn
